# ============================================
# NGINX PROXY MANAGER - Advanced Configuration
# ============================================
# Copy this into the "Advanced" tab when creating the Proxy Host
# in Nginx Proxy Manager for classroompath.duckdns.org

# API routes → forward to API container (port 3000)
location /api/ {
    proxy_pass http://classroompath-api:3000;
    proxy_http_version 1.1;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
}

location /trpc/ {
    proxy_pass http://classroompath-api:3000;
    proxy_http_version 1.1;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
}

location /health {
    proxy_pass http://classroompath-api:3000;
    proxy_http_version 1.1;
    proxy_set_header Host $host;
}

location /w/ {
    proxy_pass http://classroompath-api:3000;
    proxy_http_version 1.1;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
}

# SPA (everything else) → forward to SPA container (port 8080)
location / {
    proxy_pass http://classroompath-spa:80;
    proxy_http_version 1.1;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
}
